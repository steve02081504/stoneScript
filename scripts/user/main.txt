var speedRun = false // true=竞速：不采集、不图鉴、超时自动 Leave
var useAutoPotion = true // false=关闭 loop 末尾基础自动药水

var finisherWeapon = "" // 终结武器 equip id；空=不启用
var finisherCooldown = "" // cooldown id，与上行同填才启用
var finisherLastHitOnly = false // true=仅残血时激活终结技R

var bq = [["aether","vigor"],["fire","aether"],["vigor","poison"],["poison","ice"],["ice","fire"]]
var bd = [["damage","morel",0.25],["damage","yeti",0.15],["damage","rocky",0],["chill","morel",0.1],["chill","rocky",0.1],["dot","morel",0.1],["dot","yeti",0.15],["dot","pallas",0.05],["dot","rocky",0.05],["feeble","shroom",0.15],["feeble","rocky",0.15]]
func fe()
  ?item.left.state = 3
    ?item.left ! sight
      equipL stone throwing
      equipL @item.left@
  ?item.right.state = 3
    ?item.right ! sight
      equipR shield *0*
      equip @item.right@
func ab()
  return loc = temple & foe = acronian
func eq()
  ?loc = cave | loc = temple
    return "ice"
  ?loc = fungus | loc = forest | loc = shroom
    return "poison"
  ?loc = crypt | loc = halls | loc = haunted | loc = hall | loc = undead
    return "vigor"
  ?loc = mine | loc = bronze
    return "aether"
  ?loc = icy
    return "fire"
  return "vigor"
var as = false
var ar = true
var dd = -1
var de = -1
var bz = -1
var q = -1
func aw()
  bz = -1
  q = -1
  dd = -1
  de = -1
func bj()
  return foe.distance <= 10
func bi()
  return foe.distance > 10
func u()
  return foe.distance < 11
func ce()
  return foe.distance > 17
func fi()
  return foe = undamageable | (foe.hp <= 0 & foe.armor <= 0)
func eg()
  return foe.state = 0 | fi()
func ei()
  return foe = flying | foe = mosquito
func cs()
  return foe = immune_to_ranged | eg()
func et()
  return !eg() & foe.state > 1
func ff(a)
  return item.GetCooldown(a) <= 0 & item.CanActivate()
func db()
  ?item.right ! empty & item.right.state = 2
    return true
  ?item.left ! empty & item.left.state = 2
    return true
  return false
func fx(a)
  ?item.right = @a@ & item.right.state = 2
    activate R
    return true
  return false
func fa(a)
  return a >= 0 & totaltime < a
func ds(a, b, c)
  ?!c & foe = boss
    return false
  ?foe.count < a
    return false
  ?foe = spawner
    return false
  ?b > 0 & foe.distance > b
    return false
  ?item.GetCount("blade of the fallen god") <= 0 & item.right ! blade of the fallen god
    return false
  ?!ff("blade")
    return false
  return true
func v()
  ?loc.averageTime <= 0
    return false
  var fc = (loc.averageTime - totaltime) / 30
  return fc <= 120 & fc >= 90
func cr(a, b, c, d)
  ?!ds(a, b, c)
    return false
  ?!d & v()
    return false
  return true
func em(a, b, c, d)
  ?fx("blade of the fallen god")
    ?eg() | bi()
      return false
    return true
  ?!cr(a, b, c, d)
    return false
  equip blade of the fallen god
  activate R
  return true
func es(a, b)
  var ek = string.Split(b, ",")
  for buffStr : ek
    var ex = string.Split(buffStr, ":")
    ?Type(ex) = Type([]) & ex.Count() = 4
      ?string.IndexOf(ex[1], a) > -1
        return int.Parse(ex[2])
  return 0
func fl()
  ?maxhp <= 0
    return 1
  return (hp) / (maxhp)
func fk(a)
  return fl() < a
func fj(a)
  return fl() > a
func da()
  ?foe.maxhp + foe.maxarmor <= 0
    return 1
  return ((foe.hp + foe.armor)) / ((foe.maxhp + foe.maxarmor))
func bh(a)
  dd = totaltime + 60
  equip @a@
func cx(a)
  ?item.GetCount("mind") <= 0
    return false
  equipL mind
  dd = totaltime + 60
  ?a
    de = totaltime + 60
  return true
func dk()
  ?item.GetCount("triskelion") > 0
    equipL triskelion
    equipR shield
    return true
  ?item.GetCount("shield") > 0
    equipR shield
    return true
  return false
func cd()
  equipL hammer
  equipR shield
func bf(a)
  ?cw()
    ?a = ""
      equipL hammer
    :
      equipL @a + " hammer"@
    return true
  return false
func fg(a)
  equipL @a + " hammer"@
  equipR @a + " shield"@
func fv(a)
  ?cs()
    fg(a)
    return
  bh(a + " crossbow")
func fh(a)
  bh(a + " staff")
func ac(a)
  equipR @a + " wand"@
  equipL @a + " shield"@
func fs(a)
  var fo = "dI " + a + " sword"
  var fp = "dP " + a + " sword"
  ?item.GetCount(@fp@) > 0 & item.GetCount(@fo@) > 0
    equipL dP @a + " sword"@
    equipR dI @a + " sword"@
    return true
  ez(a)
func fy(a)
  var fo = "dI " + a + " sword"
  ?item.GetCount(@fo@) > 0
    equipL @a + " sword"@
    equipR dI @a + " sword"@
    return true
  fs(a)
func ft(a)
  equipL @a + " sword"@
  equipR @a + " sword"@
func ez(a)
  ft(a)
  var ga = a + " shield"
  ?item.GetCount(@ga@) > 0
    equipR @ga@
  return true
func j()
  ?foe.distance > 17 & !cs()
    equip vigor crossbow
    return
  equipL vigor sword
  equipR vigor sword
func dm(a, b)
  ?a < 0
    return true
  ?b = 1
    return foe.time = a
  ?b = 0
    return foe.time >= a
  ?b = 2
    return foe.time > a
  return true
func fu(a, b, c, d, e)
  ?foe.a ! a
    return false
  ?!dm(b, c)
    return false
  ?d & item.GetCooldown("mind") > 0
    return false
  ?cx(e)
    return true
  return false
func cw()
  ?item.GetCount("bashing shield") <= 0
    return false
  ?ff("bash")
    equipR bashing shield
    return true
  return false
func cy()
  ?armor >= 14
    return false
  ?item.GetCount("compound shield") > 0
    equipR compound shield
    return true
  return false
func af()
  ?fx("heavy hammer")
    return true
  ?item.GetCount("heavy hammer") <= 0 & item.right ! heavy hammer
    return false
  ?bi()
    return false
  ?item.right = heavy hammer & item.GetCooldown("heavy_hammer") > 0
    equip heavy hammer
    return true
  ?ff("heavy_hammer")
    equip heavy hammer
    activate R
    return true
  return false
func co(a)
  ?a
    return af()
  ?item.GetCount("heavy hammer") <= 0
    return false
  equip heavy hammer
  return true
func cp()
  return fk(0.6) & foe.distance > 10 & foe.distance <= 17
func bg(a)
  ?fk(0.7)
    ac(a)
  :
    fh(a)
  return true
func ct()
  ?item.potion ! Healing & item.potion ! auto
    return false
  ?hp < 7
    activate potion
  return true
func bv()

    return false
  activate potion
  return true
func bw()
  ?item.potion ! Defensive
    return false
  ?hp < 5
    activate potion
    return true
  ?armor < 2 & fk(0.7)
    activate potion
  return true
func bm()
  return buffs.GetTime("berserk") > 0 | buffs.GetTime("lucky") > 0 | buffs.GetTime("strength") > 0
func cv(a, b)
  ?item.potion ! a | bm()
    return false
  ?b
    activate potion
    return true
  return false
func bo()
  ?item.potion ! Experience
    return false
  ?foe.count > 3 & foe ! boss & fj(0.8)
    activate potion
  return true
func ck()
  ?item.potion ! Vampiric
    return false
  ?et() & foe ! boss & fk(0.85) & hp > 5
    activate potion
  return true
func ap()
  ?item.potion ! Invisibility
    return false
  ?foe.count > 2 & fk(0.3)
    activate potion
  return true
func fm()
  ?!item.CanActivate()
    return
  ?item.potion = empty
    return
  ?ct()
    return
  ?debuffs.count > 0 | fk(0.5)
    ?bv()
      return
  ?bw()
    return
  ?cv(Strength, foe.maxarmor > 0 & foe.distance < 15)
    return
  ?bo()
    return
  ?cv(Lightning, foe.count >= 4)
    return
  ?ck()
    return
  ap()
func ep(a)
  for n : a
    ?foe = @n@
      return true
  return false
func av(a)
  ?a = "shroom"
    return ep(["bolesh", "morel", "yeti"])
  ?a = "rocky"
    return loc = rocky
  return foe = @a@
func be(a)
  for row : bd
    ?row[0] = a & av(row[1])
      return row[2]
  return 0
func r(a)
  ?ep(["poena", "guardian", "nagaraja"])
    return true
  ?a = "feeble" & ep(["pallas"])
    return true
  ?a ! "dot" & a ! "feeble" & ep(["bolesh", "pallas"])
    return true
  ?a = "dot" & ep(["bolesh"])
    return true
  return false
func s()
  return foe.buffs.string = protection | bm()
func er(a)
  return ce() | s() | (foe = legless & a ! "feeble") | dp(a)
func au(a)
  for row : [["chill", "yeti"]]
    ?row[0] = a & foe = row[1]
      return true
  return false
func dp(a)
  ?!foe = boss
    return false
  ?r(a) | au(a)
    return true
  var ey = be(a)
  ?ey > 0 & da() <= ey
    return true
  return false
func ec(a, b, c, d)
  ?er(b)
    return false
  ?d ! ""
    ?foe.debuffs.string = @d@
      return false
  ?b = "damage" & foe = immune_to_debuff_damage
    return false
  ?b = "chill"
    ?es("chill", foe.debuffs.string) >= 6
      return false
    ?foe = immune_to_debuff_chill
      return false
  equipR mask
  equipL @c + a + " sword"@
  return true
func br(a, b)
  ?foe.debuffs.string = feeble
    return false
  ?a
    ?dp("feeble")
      return false
  :
    ?er("feeble")
      return false
  ?!ff("mask")
    return false
  equipR mask
  activate R
  ?b
  return true
func cn(a)
  ?db()
    return false
  ?s()
    return false
  ?bi()
    return false
  for row : [["damage","dP","damage"],["dot","dF","dot"],["chill","dI",""]]
    ?ec(a, row[0], row[1], row[2])
      return true
  ?foe.maxarmor > 50 & foe.debuffs.string ! fatigue
    ?af()
      return true
  return br(false, true)
func dn()
  ?foe = boss | foe = magic_resist | ei()
    return false
  ?foe = immune_to_physical | foe = immune_to_ranged
    return false
  ?foe.maxarmor <= 0
    return false
  ?foe.maxarmor > foe.maxhp / 2
    return true
  ?foe.maxhp + foe.maxarmor <= 120
    return true
  return false
func di()
  for i = 0..5 - 1
    var fz = ["vigor", "aether", "fire", "ice", "poison"][i]
    var fn = ["poison", "vigor", "aether", "fire", "ice"][i]
    ?foe.buffs.string ! ("adaptive_defense_" + fn) & foe = (" " + fz + " ")
      return fn
  for i = 0..5 - 1
    ?foe = ["vigor", "aether", "fire", "ice", "poison"][i]
      return ["poison", "vigor", "aether", "fire", "ice"][i]
  return "vigor"
