// ===== 遭遇标签 =====

export func Combat_IsAcronianFight()
  return loc = temple & foe = acronian

export func Combat_IsRockyDysangelosFight()
  return loc = rocky & (foe = boss | foe = phase1 | foe = phase2 | foe = phase3)

// bardiche 群杀不用于 Boss、圣殿 Acronian、岩石 Dysangelos 多阶段
export func Combat_SkipsBardicheBurst()
  return foe = boss | Combat_IsAcronianFight() | Combat_IsRockyDysangelosFight()

// ===== 地点默认克制元素（combat 包内）=====

// 地点默认克制元素（旅行装 EqMelee 等 fallback）
func LocElement()
  // 冰系地形
  ?loc = cave | loc = temple
    return "ice"
  // 毒系 / 真菌
  ?loc = fungus | loc = forest | loc = shroom
    return "poison"
  // 亡灵 / 墓穴
  ?loc = crypt | loc = halls | loc = haunted | loc = hall | loc = undead
    return "vigor"
  // 矿洞 / 青铜
  ?loc = mine | loc = bronze
    return "aether"
  // 雪地
  ?loc = icy
    return "fire"
  return "vigor"
